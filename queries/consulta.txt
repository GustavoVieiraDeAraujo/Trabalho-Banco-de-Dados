Listar jogadores e os títulos que conquistaram em um time específico

SELECT Jogador.Nome_Jogador, Título.Nome_Título
FROM Time
JOIN Jogador ON Time.ID_Time = Jogador.ID_Time
JOIN Título ON Jogador.ID_Jogador = Título.ID_Jogador
WHERE Time.Nome_Time = 'Time_X';

Obter o estádio e as suas respectivas localizações para os jogos de uma competição

SELECT Estádio.Nome_Estádio, Localização.Nome_Localização
FROM Estádio
JOIN Localização ON Estádio.ID_Localização = Localização.ID_Localização
JOIN Jogo ON Estádio.ID_Estádio = Jogo.ID_Estádio
JOIN Competição ON Jogo.ID_Competição = Competição.ID_Competição
WHERE Competição.ID_Competição = 'X';

Listar os jogadores e seus respectivos técnicos de um time

SELECT Jogador.Nome_Jogador, Técnico.Nome_Técnico
FROM Jogador
JOIN Time ON Jogador.ID_Time = Time.ID_Time
JOIN Técnico ON Time.ID_Time = Técnico.ID_Time
WHERE Time.Nome_Time = 'Time_X';

Obter os jogos de uma competição e os times que participaram, com o estádio onde ocorreu o jogo

SELECT Competição.Nome_Competição, Jogo.Nome_Jogo, Time.Nome_Time, Estádio.Nome_Estádio
FROM Jogo
JOIN Time ON Jogo.ID_Jogo = Time.ID_Jogo
JOIN Estádio ON Jogo.ID_Estádio = Estádio.ID_Estádio
JOIN Competição ON Jogo.ID_Competição = Competição.ID_Competição
WHERE Competição.ID_Competição = 'X';
